# Series v1 Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build `series-v1/` — a horizontal ribbon of 9 particle-cloud paintings with arrow-key snap navigation and edge-bleed sandstorm transitions.

**Architecture:** One `PointsNodeMaterial` shared across 9 `Points` meshes, each built from a trimmed PNG's alpha+RGB pixels with a precomputed per-particle edge-distance. A transition state machine drives per-mesh gust uniforms and a global wind direction; only the outgoing/incoming pair animate during any snap. Ambient wave drift from pointcloud-v2 runs on all paintings at rest.

**Tech Stack:** three.js 0.170 WebGPU + TSL, lil-gui, ES module import map (same pattern as `pointcloud-v2/`).

**Design doc:** `docs/plans/2026-04-24-series-v1-design.md`

**Testing note:** This repo's prototypes (pointcloud-v*, birds-v1) have no automated tests — they're visual WebGPU pieces. Each task below ends with a **visual verification step**: open the page in a browser and confirm the described state before committing. Don't skip verification; bad shader math is silent.

```js
const PARTICLES_PER_PAINTING = 250_000;
const PAINTING_HEIGHT = 2.0;
const ALPHA_THRESHOLD = 0.5 * 255;

const PAINTINGS = [
  '0012', '0013', '0018', '0020', '0022', '0029', '0031', '0032', '0038',
].map(n => `../shared/images/series-1/17_RESAN_OCH_ORIENTEN_LL_${n}-trim.png`);

function loadImage(src) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error(`Failed: ${src}`));
    img.src = src;
  });
}

function getPixels(img, maxW = 1600) {
  let w = img.width, h = img.height;
  if (w > maxW) { const s = maxW / w; w = Math.round(w * s); h = Math.round(h * s); }
  const c = document.createElement('canvas');
  c.width = w; c.height = h;
  const ctx = c.getContext('2d');
  ctx.drawImage(img, 0, 0, w, h);
  return ctx.getImageData(0, 0, w, h);
}

// Build per-painting particle arrays. Returns { positions, colors, seeds, edgeDist, width, height }.
// width/height are world-space dimensions of the painting (aspect-preserved, height = PAINTING_HEIGHT).
function buildPaintingParticles(pixels, count) {
  const { data, width: iw, height: ih } = pixels;
  const aspect = iw / ih;
  const worldH = PAINTING_HEIGHT;
  const worldW = worldH * aspect;

  // Collect all opaque pixel indices first (for uniform sampling)
  const opaque = [];
  for (let y = 0; y < ih; y++) {
    for (let x = 0; x < iw; x++) {
      const i = (y * iw + x) * 4;
      if (data[i + 3] > ALPHA_THRESHOLD) opaque.push(y * iw + x);
    }
  }
  if (opaque.length === 0) throw new Error('No opaque pixels in painting');

  const positions = new Float32Array(count * 3);
  const colors = new Float32Array(count * 3);
  const seeds = new Float32Array(count * 4);
  // edgeDist computed in Task 8; fill with 1.0 (all "interior") for now
  const edgeDist = new Float32Array(count).fill(1.0);

  for (let p = 0; p < count; p++) {
    const pixelIdx = opaque[Math.floor(Math.random() * opaque.length)];
    const px = pixelIdx % iw;
    const py = Math.floor(pixelIdx / iw);
    const rgba = pixelIdx * 4;

    const nx = px / iw, ny = py / ih;
    // sub-pixel jitter so particles don't grid-align
    const jx = (Math.random() - 0.5) * (worldW / iw);
    const jy = (Math.random() - 0.5) * (worldH / ih);
    positions[p * 3]     = (nx - 0.5) * worldW + jx;
    positions[p * 3 + 1] = (0.5 - ny) * worldH + jy;
    positions[p * 3 + 2] = (Math.random() - 0.5) * 0.16;

    colors[p * 3]     = data[rgba]     / 255;
    colors[p * 3 + 1] = data[rgba + 1] / 255;
    colors[p * 3 + 2] = data[rgba + 2] / 255;

    seeds[p * 4]     = (Math.random() - 0.5) * 2;
    seeds[p * 4 + 1] = (Math.random() - 0.5) * 2;
    seeds[p * 4 + 2] = Math.random();
    seeds[p * 4 + 3] = 0.3 + Math.random() * 0.7;
  }

  return { positions, colors, seeds, edgeDist, worldW, worldH };
}
```

```js
function easeInOut(t) { return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2; }

  if (transitioning) return;
  const clamped = Math.max(0, Math.min(PAINTINGS.length - 1, i));
  if (clamped === currentIndex) return;

  if (e.key === 'ArrowRight') snapTo(currentIndex + 1);
  if (e.key === 'ArrowLeft')  snapTo(currentIndex - 1);
  if (e.key === 'f' || e.key === 'F') {
    if (!document.fullscreenElement) document.documentElement.requestFullscreen().catch(() => {});
```

## Task 8: Precompute per-particle edge-distance

The trim PNG's alpha defines the painting shape. For each particle, compute the distance (in pixels) to the nearest transparent pixel, normalized to 0..1. This drives the edge-bleed in Task 9.

**Files:**
- Modify: `series-v1/main.js`

**Step 1: Add a 2-pass sweep distance transform**

Above `buildPaintingParticles`, add:

```js
// Chamfer distance transform on alpha. Returns Float32Array of length iw*ih:
// 0 at transparent pixels, larger values deeper inside. Not true Euclidean, but
// close enough for a soft edge-band.
function alphaDistanceField(pixels) {
  const { data, width: iw, height: ih } = pixels;
  const INF = 1e9;
  const d = new Float32Array(iw * ih);
  for (let i = 0; i < d.length; i++) {
    d[i] = data[i * 4 + 3] > ALPHA_THRESHOLD ? INF : 0;
  }
  // Forward pass
  for (let y = 0; y < ih; y++) {
    for (let x = 0; x < iw; x++) {
      const i = y * iw + x;
      if (d[i] === 0) continue;
      if (x > 0)           d[i] = Math.min(d[i], d[i - 1] + 1);
      if (y > 0)           d[i] = Math.min(d[i], d[i - iw] + 1);
      if (x > 0 && y > 0)  d[i] = Math.min(d[i], d[i - iw - 1] + 1.41421);
      if (x < iw-1 && y>0) d[i] = Math.min(d[i], d[i - iw + 1] + 1.41421);
    }
  }
  // Backward pass
  for (let y = ih - 1; y >= 0; y--) {
    for (let x = iw - 1; x >= 0; x--) {
      const i = y * iw + x;
      if (d[i] === 0) continue;
      if (x < iw - 1)           d[i] = Math.min(d[i], d[i + 1] + 1);
      if (y < ih - 1)           d[i] = Math.min(d[i], d[i + iw] + 1);
      if (x < iw-1 && y < ih-1) d[i] = Math.min(d[i], d[i + iw + 1] + 1.41421);
      if (x > 0 && y < ih - 1)  d[i] = Math.min(d[i], d[i + iw - 1] + 1.41421);
    }
  }
  return d;
}
```

```js
    const w1z = sin(wave1Phase.mul(0.5).add(1.0)).mul(uWaveAmp.mul(0.6));
    const wave2Phase = pos.y.mul(uWave2Freq).add(pos.z.mul(2.0)).add(t.mul(uWave2Spd));
    const w2x = cos(wave2Phase).mul(uWave2Amp.mul(0.5));
    const w2y = sin(wave2Phase).mul(uWave2Amp);
    const w2z = cos(wave2Phase.mul(1.3)).mul(uWave2Amp.mul(0.7));

    const dy = cos(t.mul(0.11).add(seedY.mul(5.0))).mul(uMicroDrift);

    return pos.add(vec3(
      w1x.add(w2x).add(dx),
      w1y.add(w2y).add(dy),
      w1z.add(w2z),
    ));
  })();
  return m;
}
```

```js

// travelCurve: stride-scale rightward drift for outgoing, leftward pull for incoming
// For an outgoing particle: wind pushes in +windDir * stride * detach.
// We don't yet distinguish outgoing vs incoming — both paintings' detached particles
// drift in the wind direction. This is intentional: outgoing particles blow out,
// incoming particles arrive having just blown in.

  w1z.add(w2z).add(windZ),
```

```js
function snapTo(i) {
```
